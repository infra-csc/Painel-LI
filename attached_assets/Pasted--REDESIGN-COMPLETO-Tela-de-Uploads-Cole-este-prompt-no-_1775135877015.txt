PARTE 4 — CORRIGIR O PROCESSAMENTO QUE TRAVA
```typescript
// PROBLEMA: upload fica "Aguardando" e não processa
// CAUSA PROVÁVEL: processamento síncrono na request HTTP
// que excede o timeout do servidor

// SOLUÇÃO: processar em background com polling de progresso

// 1. Rota de upload — apenas recebe o arquivo e inicia job:
POST /api/uploads/horas
→ Salva arquivo no storage temporário
→ Cria registro na tabela uploads com status 'PROCESSANDO'
→ Dispara processamento assíncrono (não await)
→ Retorna imediatamente: { uploadId, status: 'PROCESSANDO' }

// 2. Processamento em background:
async function processarCSVBackground(uploadId: string, filePath: string) {
  try {
    // FASE 1: Limpeza
    await atualizarProgresso(uploadId, 'limpando', 0)
    await limparDadosAnteriores()
    await atualizarProgresso(uploadId, 'limpando', 100)

    // FASE 2: Processar registros em lotes de 1000
    await atualizarProgresso(uploadId, 'processando', 0)
    const linhas = await lerCSV(filePath)
    for (let i = 0; i < linhas.length; i += 1000) {
      await processarLote(linhas.slice(i, i + 1000), uploadId)
      await atualizarProgresso(
        uploadId, 'processando',
        Math.round((i / linhas.length) * 100)
      )
    }

    // FASE 3: Calcular turnos
    await atualizarProgresso(uploadId, 'calculando', 0)
    await calcularTurnos(uploadId)
    await atualizarProgresso(uploadId, 'calculando', 100)

    // Concluído
    await db.query(
      `UPDATE uploads SET status = 'CONCLUIDO',
       registros = $1, periodo_inicio = $2, periodo_fim = $3
       WHERE id = $4`,
      [totalRegistros, periodoInicio, periodoFim, uploadId]
    )
  } catch (err) {
    await db.query(
      `UPDATE uploads SET status = 'ERRO', erro_mensagem = $1 WHERE id = $2`,
      [err.message, uploadId]
    )
  }
}

// 3. Rota de progresso (frontend faz polling a cada 2s):
GET /api/uploads/[id]/progresso
→ {
    fase: 'limpando' | 'processando' | 'calculando' | 'concluido' | 'erro',
    percentual: 65,
    registros_processados: 74832,
    total_registros: 114988,
    erro_mensagem: null
  }

// 4. Frontend faz polling:
useEffect(() => {
  if (!uploadEmAndamento) return
  const interval = setInterval(async () => {
    const progresso = await fetch(`/api/uploads/${uploadId}/progresso`)
    setProgresso(progresso)
    if (['concluido', 'erro'].includes(progresso.fase)) {
      clearInterval(interval)
      setUploadEmAndamento(false)
    }
  }, 2000) // a cada 2 segundos
  return () => clearInterval(interval)
}, [uploadEmAndamento, uploadId])
```

PARTE 6 — TABELA uploads NO BANCO
```sql
-- Garantir campos necessários:
ALTER TABLE uploads
  ADD COLUMN IF NOT EXISTS tipo          VARCHAR(20) DEFAULT 'HORAS',
  -- 'COLABORADORES' ou 'HORAS'
  ADD COLUMN IF NOT EXISTS fase          VARCHAR(20),
  -- 'limpando', 'processando', 'calculando'
  ADD COLUMN IF NOT EXISTS percentual    INTEGER DEFAULT 0,
  ADD COLUMN IF NOT EXISTS registros_processados INTEGER DEFAULT 0,
  ADD COLUMN IF NOT EXISTS total_registros       INTEGER DEFAULT 0,
  ADD COLUMN IF NOT EXISTS erro_mensagem TEXT,
  ADD COLUMN IF NOT EXISTS periodo_inicio DATE,
  ADD COLUMN IF NOT EXISTS periodo_fim    DATE,
  ADD COLUMN IF NOT EXISTS colaboradores  INTEGER DEFAULT 0;

-- Função para atualizar progresso:
CREATE OR REPLACE FUNCTION atualizar_progresso(
  p_id UUID, p_fase VARCHAR, p_percentual INTEGER,
  p_processados INTEGER DEFAULT NULL
) RETURNS void AS $$
BEGIN
  UPDATE uploads SET
    fase = p_fase,
    percentual = p_percentual,
    registros_processados = COALESCE(p_processados, registros_processados)
  WHERE id = p_id;
END;
$$ LANGUAGE plpgsql;
```
